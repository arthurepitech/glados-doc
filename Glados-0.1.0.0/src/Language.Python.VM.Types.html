-- Instructions de la machine virtuelle
data VMInstruction
    = LOAD_CONST PyValue    -- Charge une constante sur la pile
    | LOAD_VAR String       -- Charge une variable sur la pile
    | STORE_VAR String      -- Stocke une valeur dans une variable
    | BINARY_OP String      -- Opération binaire (+, -, *, /, **)
    | UNARY_OP String       -- Opération unaire (-)
    | CALL String Int       -- Appel de fonction (nom, nombre d'arguments)
    | RETURN                -- Retour de fonction
    | JUMP Int              -- Saut inconditionnel
    | JUMP_IF_FALSE Int     -- Saut conditionnel
    | MAKE_LIST Int         -- Crée une liste avec n éléments
    | MAKE_FUNCTION         -- Crée une fonction (pas encore implémenté)
    | POP                   -- Dépile le sommet de la pile
    deriving (Int -> VMInstruction -> ShowS
